public enum RouteDiscipline { Pipe, Conduit }

public sealed class RouteRequest
{
    public Connector Source { get; init; }
    public Connector Target { get; init; }
    public IRoutingBoundary Boundary { get; init; }
    public RoutingOptions Options { get; init; }
    public ElementId PipeTypeId { get; init; }
    public ElementId SystemTypeId { get; init; }
    public ElementId LevelId { get; init; }
    public RouteDiscipline Discipline { get; init; }
    public bool Commit { get; init; } = true;

    // [v4-1]
    public PipeSizeInfo? Size { get; init; }                         // null = match source connector
    public double MinSegmentMm { get; init; } = 0;                   // 0 = 管徑 × 2
    public List<RouteProblem> PreChecks { get; init; } = new();

    // [v4.1-1]
    public bool SuppressWarnings { get; init; } = true;
    public bool AddFittings { get; init; } = true;
    public bool ConnectEnds { get; init; } = true;
}

public sealed class RouteResult
{
    public List<XYZ> Path { get; set; }
    public List<RouteProblem> Problems { get; } = new();
    public int CreatedSegments { get; set; }
    public int CreatedElbows { get; set; }
    public bool Committed { get; set; }
}

public static RouteResult Run(UIDocument uidoc, RouteRequest req, Action<string> log)
{
    var res = new RouteResult();
    res.Problems.AddRange(req.PreChecks);
    if (PathValidator.HasErrors(res.Problems)) return res;

    // [v4.1-4]
    if (req.Source == null || req.Target == null)
    { res.Problems.Add(RouteProblem.Error("Source or target connector is missing.")); return res; }
    if (req.PipeTypeId == null || req.PipeTypeId == ElementId.InvalidElementId)
    { res.Problems.Add(RouteProblem.Error("No pipe / conduit type is selected.")); return res; }
    if (req.Discipline == RouteDiscipline.Pipe && (req.SystemTypeId == null || req.SystemTypeId == ElementId.InvalidElementId))
    { res.Problems.Add(RouteProblem.Error("No piping system type is selected.")); return res; }
    if (req.LevelId == null || req.LevelId == ElementId.InvalidElementId)
    { res.Problems.Add(RouteProblem.Error("No reference level is selected.")); return res; }

    Document doc = uidoc.Document;
    RoutingOptions o = req.Options;

    double lead = Geom.MmToFt(o.LeadLengthMm);
    var start = ConnectorEndpoint.From(req.Source, lead);
    var end   = ConnectorEndpoint.From(req.Target, lead);
    if (!start.IsAxisAligned) res.Problems.Add(RouteProblem.Warn("Source connector is not axis-aligned – snapped to the nearest axis."));
    if (!end.IsAxisAligned)   res.Problems.Add(RouteProblem.Warn("Target connector is not axis-aligned – snapped to the nearest axis."));

    // [v4-2]
    double pipeRadius = req.Size?.OuterRadiusFt ?? start.Radius;
    double diameterFt = req.Size?.NominalFt ?? start.Radius * 2;
    double routeDiaMm = Geom.FtToMm(diameterFt);
    foreach (var (ep, label) in new[] { (start, "Source"), (end, "Target") })
    {
        double epMm = Geom.FtToMm(ep.Radius * 2);
        if (Math.Abs(epMm - routeDiaMm) > 0.5)
            res.Problems.Add(RouteProblem.Warn(
                $"{label} connector is Ø{epMm:0} mm but the route is Ø{routeDiaMm:0} mm – a reducer is required."));
    }
    log?.Invoke($"Start · route Ø{routeDiaMm:0} mm · lead {o.LeadLengthMm:0} mm");            // [v4.1-2]

    double cell = CellSizeFt;
    double clearance = Geom.MmToFt(o.WallClearanceMm);
    if (clearance > 0 && cell > clearance)
        res.Problems.Add(RouteProblem.Warn(
            $"Cell size {Geom.FtToMm(cell):0} mm is larger than the wall clearance {o.WallClearanceMm:0} mm – clearance may be inaccurate."));

    IRoutingBoundary b = req.Boundary;
    if (b != null && (!b.Contains(start.Lead) || !b.Contains(end.Lead)))
    {
        res.Problems.Add(RouteProblem.Error($"Source or target is outside {b.Kind} '{b.Name}'."));
        return res;
    }

    // ... [現有 region code] ...
    log?.Invoke($"Region {Geom.FtToMm(regionMax.X - regionMin.X):0} × {Geom.FtToMm(regionMax.Y - regionMin.Y):0} × {Geom.FtToMm(regionMax.Z - regionMin.Z):0} mm" +
                (b != null ? $" ({b.Kind} {b.Name})" : " (default volume)"));

    // ... [現有 walls / constraints / grid / BlockSolid / UnblockCorridor code] ...
    log?.Invoke($"Grid {grid.NX} × {grid.NY} × {grid.NZ} = {(long)grid.NX * grid.NY * grid.NZ:N0} cells @ {Geom.FtToMm(cell):0} mm");
    log?.Invoke($"Grid ready · {blocked} cells blocked by constraints · {walls.Count} walls");

    // ... [A*] ...
    log?.Invoke($"A* finished · {mid?.Count ?? 0} points");
    if (!PathUtils.IsValid(mid))
    {
        res.Problems.Add(RouteProblem.Error(
            "No path found. Try a smaller wall clearance, a larger boundary or a shorter lead length."));
        return res;
    }

    // ... [full path / MergeCollinear / SlopeApplier] ...
    log?.Invoke($"Path {path.Count} points · slope {o.Slope.Display} · fall {Geom.FtToMm(SlopeApplier.TotalFall(path)):0} mm");

    // [v4-3]
    double minSegFt = req.MinSegmentMm > 0 ? Geom.MmToFt(req.MinSegmentMm) : diameterFt * 2;
    res.Problems.AddRange(PathValidator.Validate(path, minSegFt, p => grid.IsBlockedWorld(p), cell / 2));
    res.Path = path;

    if (!req.Commit) return res;
    if (PathValidator.HasErrors(res.Problems)) { log?.Invoke("Errors found – nothing was created."); return res; }

    // ---- [v4.1-3] Commit ----
    log?.Invoke($"Creating {path.Count - 1} segments…");
    var failures = new List<string>();
    using (var tx = new Transaction(doc, "MEP Auto Route"))
    {
        tx.Start();
        RouteFailureCollector fc = RouteFailureCollector.Attach(tx, req.SuppressWarnings);

        var segs = new List<MEPCurve>();
        try
        {
            for (int i = 0; i < path.Count - 1; i++)
            {
                MEPCurve c = req.Discipline == RouteDiscipline.Pipe
                    ? Pipe.Create(doc, req.SystemTypeId, req.PipeTypeId, req.LevelId, path[i], path[i + 1])
                    : Conduit.Create(doc, req.PipeTypeId, path[i], path[i + 1], req.LevelId);

                BuiltInParameter dp = req.Discipline == RouteDiscipline.Pipe
                    ? BuiltInParameter.RBS_PIPE_DIAMETER_PARAM : BuiltInParameter.RBS_CONDUIT_DIAMETER_PARAM;
                c.get_Parameter(dp)?.Set(diameterFt);
                segs.Add(c);
            }
        }
        catch (Exception ex)
        {
            tx.RollBack();
            res.Problems.Add(RouteProblem.Error(
                $"Segment {segs.Count} could not be created – {ex.GetType().Name}: {ex.Message}"));
            return res;
        }
        log?.Invoke($"{segs.Count} segments created, adding elbows…");

        if (req.AddFittings) res.CreatedElbows = MepFittingUtils.CreateElbows(doc, segs, path, failures);
        if (req.ConnectEnds)
        {
            TryConnect(segs[0], start.Connector, start.Origin, failures);
            TryConnect(segs[^1], end.Connector, end.Origin, failures);
        }

        TransactionStatus st = tx.Commit();

        foreach (string w in fc.Warnings) res.Problems.Add(RouteProblem.Warn($"Revit: {w}"));
        foreach (string e in fc.Errors)   res.Problems.Add(RouteProblem.Error($"Revit: {e}"));

        if (st != TransactionStatus.Committed)
        {
            res.Problems.Add(RouteProblem.Error(
                $"Revit rolled back the transaction ({st}) – nothing was created. See the Revit errors above."));
            return res;
        }

        res.CreatedSegments = segs.Count;
        res.Committed = true;
    }
    foreach (string f in failures) res.Problems.Add(RouteProblem.Warn(f));
    log?.Invoke($"Created {res.CreatedSegments} segments and {res.CreatedElbows} elbows · size Ø{routeDiaMm:0} mm · slope {o.Slope.Display}");
    return res;
}

private static void TryConnect(MEPCurve seg, Connector device, XYZ at, List<string> failures)
{
    Connector c = MepFittingUtils.ClosestEndConnector(seg, at);
    if (c == null || device == null || device.IsConnected) return;
    try { c.ConnectTo(device); }
    catch (Exception ex) { failures.Add($"Could not connect to the equipment connector: {ex.Message}"); }
}
